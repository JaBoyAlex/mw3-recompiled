cmake_minimum_required(VERSION 3.25)
project(mw3_memory_contract LANGUAGES CXX)
# Isolated SDK-facing test adapter; public core remains C++20.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rexglue 0.11 REQUIRED CONFIG)
foreach(required PPC_IMAGE_BASE PPC_IMAGE_SIZE PPC_CODE_BASE PPC_CODE_SIZE)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing validated synthetic/title geometry: ${required}")
  endif()
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/geometry.h" "#pragma once\n#define PPC_CONFIG_H_INCLUDED\n#define PPC_IMAGE_BASE ${PPC_IMAGE_BASE}ull\n#define PPC_IMAGE_SIZE ${PPC_IMAGE_SIZE}ull\n#define PPC_CODE_BASE ${PPC_CODE_BASE}ull\n#define PPC_CODE_SIZE ${PPC_CODE_SIZE}ull\n")
add_executable(memory-contract contract.cpp)
target_include_directories(memory-contract PRIVATE "${CMAKE_BINARY_DIR}")
target_compile_options(memory-contract PRIVATE -include "${CMAKE_BINARY_DIR}/geometry.h")
option(MW3_MEMORY_SANITIZERS "Instrument original adapter, not the packaged SDK" OFF)
if(MW3_MEMORY_SANITIZERS)
  if(WIN32)
    target_compile_options(memory-contract PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
    target_link_options(memory-contract PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
  else()
    target_compile_options(memory-contract PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(memory-contract PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
  endif()
endif()
target_link_libraries(memory-contract PRIVATE rex::sdk)
rexglue_apply_target_settings(memory-contract)
rexglue_configure_target(memory-contract)

option(MW3_UPSTREAM_MEMORY_TESTS "Compile focused pinned upstream tests against SDK DLLs" OFF)
if(MW3_UPSTREAM_MEMORY_TESTS)
  if(NOT DEFINED MW3_SDK_SOURCE)
    message(FATAL_ERROR "Missing verified SDK test/Catch2 source")
  endif()
  set(CATCH_INSTALL_DOCS OFF CACHE BOOL "" FORCE)
  set(CATCH_INSTALL_EXTRAS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${MW3_SDK_SOURCE}/thirdparty/catch2" "${CMAKE_BINARY_DIR}/catch2" EXCLUDE_FROM_ALL)
  # Do not reuse unit_tests: upstream embeds kernel objects in that executable.
  # These original test sources call the actual packaged kernel DLL instead.
  add_executable(upstream-memory-tests
    "${MW3_SDK_SOURCE}/tests/unit/test_memory.cpp"
    "${MW3_SDK_SOURCE}/tests/unit/kernel/heap_allocation_test.cpp"
    "${MW3_SDK_SOURCE}/tests/unit/kernel/access_violation_test.cpp")
  target_include_directories(upstream-memory-tests PRIVATE "${MW3_SDK_SOURCE}/tests" "${MW3_SDK_SOURCE}/tests/unit")
  target_link_libraries(upstream-memory-tests PRIVATE rex::sdk Catch2::Catch2WithMain)
  rexglue_apply_target_settings(upstream-memory-tests)
  rexglue_configure_target(upstream-memory-tests)
endif()
