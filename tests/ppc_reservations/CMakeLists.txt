cmake_minimum_required(VERSION 3.25)
project(mw3_ppc_reservations LANGUAGES CXX)
# SDK-facing diagnostic only; pinned headers require C++23.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rexglue 0.11 REQUIRED CONFIG)
foreach(required MW3_GENERATED MW3_JSON_INCLUDE)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing verified input ${required}")
  endif()
endforeach()
add_library(reservation-guest OBJECT "${MW3_GENERATED}/ppc_test_functions.cpp")
target_compile_options(reservation-guest PRIVATE -fwrapv -ffp-contract=off)
add_executable(ppc-reservations runner.cpp)
target_link_libraries(ppc-reservations PRIVATE reservation-guest)
foreach(target reservation-guest ppc-reservations)
  target_include_directories(${target} PRIVATE "${MW3_GENERATED}" "${MW3_JSON_INCLUDE}")
  target_link_libraries(${target} PRIVATE rex::sdk)
  rexglue_apply_target_settings(${target})
endforeach()
rexglue_configure_target(ppc-reservations)
option(MW3_RESERVATION_SANITIZERS "Instrument runner and generated instructions" OFF)
if(MW3_RESERVATION_SANITIZERS)
  foreach(target reservation-guest ppc-reservations)
    if(WIN32)
      target_compile_options(${target} PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
      target_link_options(${target} PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
    else()
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
    endif()
  endforeach()
endif()
