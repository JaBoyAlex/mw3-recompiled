cmake_minimum_required(VERSION 3.25)
project(mw3_audio_lifetimes LANGUAGES CXX)
enable_testing()
get_filename_component(MW3_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(MW3_AUDIO_CORE "${MW3_ROOT}/runtime/audio/streams" CACHE PATH "Original maintained core, or private snapshot")
option(MW3_AUDIO_SDK_PROBE "Build the separately verified actual-SDK diagnostic" OFF)
option(MW3_AUDIO_STREAM_SANITIZERS "Instrument maintained audio ownership code" OFF)
add_library(mw3_audio_streams STATIC "${MW3_AUDIO_CORE}/streams.cpp")
target_include_directories(mw3_audio_streams PUBLIC "${MW3_AUDIO_CORE}")
set_target_properties(mw3_audio_streams PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(MW3_AUDIO_SDK_PROBE)
  find_package(rexglue 0.11 REQUIRED CONFIG)
  find_package(SDL3 REQUIRED CONFIG)
  add_executable(audio-streams-sdk sdk_contracts.cpp)
  target_link_libraries(audio-streams-sdk PRIVATE mw3_audio_streams rex::sdk SDL3::SDL3 ${CMAKE_DL_LIBS})
  set_target_properties(audio-streams-sdk PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  rexglue_apply_target_settings(audio-streams-sdk)
  rexglue_configure_target(audio-streams-sdk)
  set(MW3_AUDIO_EXECUTABLE audio-streams-sdk)
else()
  add_executable(audio-streams-tests contracts.cpp)
  target_link_libraries(audio-streams-tests PRIVATE mw3_audio_streams)
  set_target_properties(audio-streams-tests PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  set(MW3_AUDIO_EXECUTABLE audio-streams-tests)
  foreach(case lifetime owners failures credits limits frames threads shutdown reentrancy)
    add_test(NAME audio-streams-${case} COMMAND audio-streams-tests ${case})
    set_tests_properties(audio-streams-${case} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
foreach(target mw3_audio_streams ${MW3_AUDIO_EXECUTABLE})
  if(MW3_AUDIO_SDK_PROBE)
    if(WIN32)
      target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
      if(MW3_AUDIO_STREAM_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
        target_link_options(${target} PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
      endif()
    elseif(MW3_AUDIO_STREAM_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
    endif()
  elseif(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    if(MW3_AUDIO_STREAM_SANITIZERS)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(MW3_AUDIO_STREAM_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
