cmake_minimum_required(VERSION 3.25)
project(mw3_sdl_trigger_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SDL3 3.4.14 EXACT REQUIRED CONFIG)
add_executable(sdl-trigger-probe probe.cpp)
target_link_libraries(sdl-trigger-probe PRIVATE SDL3::SDL3 ${CMAKE_DL_LIBS})
option(MW3_SDL_SANITIZERS "Instrument the original probe" OFF)
if(MW3_SDL_SANITIZERS)
  target_compile_options(sdl-trigger-probe PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sdl-trigger-probe PRIVATE -fsanitize=address,undefined)
endif()
