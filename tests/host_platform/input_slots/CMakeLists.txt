cmake_minimum_required(VERSION 3.25)
project(mw3_input_contracts LANGUAGES CXX)
enable_testing()
get_filename_component(MW3_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(mw3_input_slots STATIC "${MW3_ROOT}/runtime/input/slot_router.cpp")
target_include_directories(mw3_input_slots PUBLIC "${MW3_ROOT}/runtime")
target_compile_features(mw3_input_slots PUBLIC cxx_std_20)
set_target_properties(mw3_input_slots PROPERTIES CXX_EXTENSIONS OFF)
add_executable(slot-router-tests contracts.cpp)
target_link_libraries(slot-router-tests PRIVATE mw3_input_slots)
set_target_properties(slot-router-tests PROPERTIES CXX_EXTENSIONS OFF)
option(MW3_INPUT_SANITIZERS "Instrument original host input contracts" OFF)
foreach(target mw3_input_slots slot-router-tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    if(MW3_INPUT_SANITIZERS)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(MW3_INPUT_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
foreach(case four-slots assignment-guards disconnect-reconnect stale-sessions frame-validation sequence-limits event-orders)
  add_test(NAME input-${case} COMMAND slot-router-tests ${case})
  set_tests_properties(input-${case} PROPERTIES TIMEOUT 30)
endforeach()
