cmake_minimum_required(VERSION 3.25)
project(mw3_input_conversion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(input-conversion-tests conversion.cpp)
target_include_directories(input-conversion-tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../runtime")
option(MW3_INPUT_SANITIZERS "Instrument game-free conversion contracts" OFF)
if(MW3_INPUT_SANITIZERS)
  if(MSVC)
    target_compile_options(input-conversion-tests PRIVATE /fsanitize=address)
    target_link_options(input-conversion-tests PRIVATE /INCREMENTAL:NO)
  else()
    target_compile_options(input-conversion-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(input-conversion-tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
foreach(case axes triggers invalid exhaustive)
  add_test(NAME ${case} COMMAND input-conversion-tests ${case})
endforeach()
