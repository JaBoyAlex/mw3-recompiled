cmake_minimum_required(VERSION 3.25)
project(mw3_sdl_backend_contracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SDL3 3.4.14 EXACT REQUIRED CONFIG)
add_executable(input-backend-contracts contracts.cpp runtime/input/backend/backend.cpp
  runtime/input/ingress/ingress.cpp runtime/input/slot_router.cpp)
target_include_directories(input-backend-contracts PRIVATE "${CMAKE_CURRENT_LIST_DIR}/runtime")
target_link_libraries(input-backend-contracts PRIVATE SDL3::SDL3 ${CMAKE_DL_LIBS})
option(MW3_INPUT_SANITIZERS "Instrument adapter and original tests, not native SDL" OFF)
if(MW3_INPUT_SANITIZERS)
  if(WIN32)
    target_compile_options(input-backend-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all)
    target_link_options(input-backend-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all)
  else()
    target_compile_options(input-backend-contracts PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(input-backend-contracts PRIVATE -fsanitize=address,undefined)
  endif()
endif()
