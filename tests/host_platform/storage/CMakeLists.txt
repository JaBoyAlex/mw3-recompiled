cmake_minimum_required(VERSION 3.25)
project(mw3_profile_contracts LANGUAGES CXX)
enable_testing()
get_filename_component(MW3_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(mw3_profile STATIC "${MW3_ROOT}/runtime/storage/profile/store.cpp")
target_include_directories(mw3_profile PUBLIC "${MW3_ROOT}/runtime")
target_compile_features(mw3_profile PUBLIC cxx_std_20)
add_executable(profile-tests contracts.cpp)
target_link_libraries(profile-tests PRIVATE mw3_profile)
option(MW3_PROFILE_SANITIZERS "Instrument original profile storage" OFF)
foreach(target mw3_profile profile-tests)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    if(MW3_PROFILE_SANITIZERS)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(MW3_PROFILE_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
foreach(case normal names roots ownership corruption rollback crossing publication limits resources links)
  add_test(NAME profile-${case} COMMAND profile-tests ${case} "${CMAKE_CURRENT_BINARY_DIR}/runs")
  set_tests_properties(profile-${case} PROPERTIES TIMEOUT 60)
endforeach()
