cmake_minimum_required(VERSION 3.25)
project(mw3_input_ingress LANGUAGES CXX)
enable_testing()
get_filename_component(MW3_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(mw3_input_ingress STATIC "${MW3_ROOT}/runtime/input/slot_router.cpp" "${MW3_ROOT}/runtime/input/ingress/ingress.cpp")
target_include_directories(mw3_input_ingress PUBLIC "${MW3_ROOT}/runtime")
target_compile_features(mw3_input_ingress PUBLIC cxx_std_20)
add_executable(input-ingress-tests contracts.cpp)
target_link_libraries(input-ingress-tests PRIVATE mw3_input_ingress)
option(MW3_INPUT_SANITIZERS "Instrument original host ingress contracts" OFF)
foreach(target mw3_input_ingress input-ingress-tests)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    if(MW3_INPUT_SANITIZERS)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(MW3_INPUT_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
foreach(case normalization explicit-joins rejected-mutations removal-reuse backend-restart sequence-limits device-budget event-orders)
  add_test(NAME ingress-${case} COMMAND input-ingress-tests ${case})
  set_tests_properties(ingress-${case} PROPERTIES TIMEOUT 30)
endforeach()
