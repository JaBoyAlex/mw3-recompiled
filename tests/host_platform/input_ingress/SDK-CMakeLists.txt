cmake_minimum_required(VERSION 3.25)
project(mw3_sdk_input_ingress LANGUAGES CXX)
# Isolated SDK adapter; the public ingress is independently built as C++20.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(rexglue 0.11 REQUIRED CONFIG)
find_package(SDL3 REQUIRED CONFIG)
add_executable(input-ingress-contracts sdk_contracts.cpp runtime/input/slot_router.cpp runtime/input/ingress/ingress.cpp)
target_include_directories(input-ingress-contracts PRIVATE "${CMAKE_CURRENT_LIST_DIR}/runtime")
target_link_libraries(input-ingress-contracts PRIVATE rex::base SDL3::SDL3 ${CMAKE_DL_LIBS})
rexglue_apply_target_settings(input-ingress-contracts)
rexglue_link_host_allocator(input-ingress-contracts)
option(MW3_INPUT_SANITIZERS "Instrument original SDK-facing ingress contracts" OFF)
if(MW3_INPUT_SANITIZERS)
  if(WIN32)
    target_compile_options(input-ingress-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
    target_link_options(input-ingress-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
  else()
    target_compile_options(input-ingress-contracts PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(input-ingress-contracts PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
  endif()
endif()
