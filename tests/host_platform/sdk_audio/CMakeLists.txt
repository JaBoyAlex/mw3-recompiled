cmake_minimum_required(VERSION 3.25)
project(mw3_sdk_audio_diagnostics LANGUAGES CXX)
# SDK-facing diagnostics require its C++23 headers; the public core stays C++20.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(rexglue 0.11 REQUIRED CONFIG)
find_package(SDL3 REQUIRED CONFIG)
add_executable(sdk-audio-contracts contracts.cpp)
target_link_libraries(sdk-audio-contracts PRIVATE rex::sdk SDL3::SDL3 ${CMAKE_DL_LIBS})
rexglue_apply_target_settings(sdk-audio-contracts)
rexglue_configure_target(sdk-audio-contracts)
if(WIN32)
  target_compile_definitions(sdk-audio-contracts PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
option(MW3_AUDIO_SANITIZERS "Instrument original host audio diagnostics" OFF)
if(MW3_AUDIO_SANITIZERS)
  if(WIN32)
    target_compile_options(sdk-audio-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
    target_link_options(sdk-audio-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
  else()
    target_compile_options(sdk-audio-contracts PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(sdk-audio-contracts PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
  endif()
endif()
