cmake_minimum_required(VERSION 3.25)
project(mw3_sdk_input_diagnostics LANGUAGES CXX)
# Isolated SDK-facing adapter, like tests/memory; maintained public core is C++20.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(rexglue 0.11 REQUIRED CONFIG)
find_package(SDL3 REQUIRED CONFIG)
add_executable(sdk-input-contracts contracts.cpp)
target_link_libraries(sdk-input-contracts PRIVATE rex::base SDL3::SDL3 ${CMAKE_DL_LIBS})
rexglue_apply_target_settings(sdk-input-contracts)
rexglue_link_host_allocator(sdk-input-contracts)
option(MW3_INPUT_SANITIZERS "Instrument original SDK-facing host contracts" OFF)
if(MW3_INPUT_SANITIZERS)
  if(WIN32)
    target_compile_options(sdk-input-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
    target_link_options(sdk-input-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
  else()
    target_compile_options(sdk-input-contracts PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(sdk-input-contracts PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
  endif()
endif()
