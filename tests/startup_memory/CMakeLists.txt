cmake_minimum_required(VERSION 3.25)
project(mw3_allocation_import_contracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rexglue 0.11 REQUIRED CONFIG)
if(NOT DEFINED MW3_SDK_SOURCE)
  message(FATAL_ERROR "Verified SDK source required for exported accounting seam")
endif()
add_executable(allocation-contracts contracts.cpp)
target_include_directories(allocation-contracts PRIVATE "${MW3_SDK_SOURCE}/src")
target_link_libraries(allocation-contracts PRIVATE rex::sdk)
rexglue_apply_target_settings(allocation-contracts)
rexglue_configure_target(allocation-contracts)
option(MW3_ALLOCATION_SANITIZERS "Instrument original callbacks/harness" OFF)
if(MW3_ALLOCATION_SANITIZERS)
  if(WIN32)
    target_compile_options(allocation-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
    target_link_options(allocation-contracts PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
  else()
    target_compile_options(allocation-contracts PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(allocation-contracts PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
  endif()
endif()
