cmake_minimum_required(VERSION 3.25)
project(mw3_display_query_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(rexglue 0.11 REQUIRED CONFIG)
if(NOT DEFINED MW3_SDK_SOURCE)
  message(FATAL_ERROR "Missing verified lifecycle-repaired SDK source")
endif()
add_executable(display-queries probe.cpp)
target_include_directories(display-queries PRIVATE "${MW3_SDK_SOURCE}/src")
target_link_libraries(display-queries PRIVATE rex::sdk ${CMAKE_DL_LIBS})
rexglue_apply_target_settings(display-queries)
rexglue_link_host_allocator(display-queries)
option(MW3_DISPLAY_SANITIZERS "Instrument original callback; SDK coverage has its own receipt" OFF)
if(MW3_DISPLAY_SANITIZERS)
  if(WIN32)
    target_compile_options(display-queries PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
    target_link_options(display-queries PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
  else()
    target_compile_options(display-queries PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(display-queries PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
  endif()
endif()
