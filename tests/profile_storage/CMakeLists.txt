cmake_minimum_required(VERSION 3.25)
project(mw3_profile_storage LANGUAGES CXX)
# The selected SDK's public headers require C++23; public project code stays C++20.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rexglue 0.11 REQUIRED CONFIG)
add_executable(profile-storage contracts.cpp)
target_link_libraries(profile-storage PRIVATE rex::sdk ${CMAKE_DL_LIBS})
rexglue_apply_target_settings(profile-storage)
rexglue_configure_target(profile-storage)
option(MW3_STORAGE_SANITIZERS "Instrument the original diagnostic host" OFF)
if(MW3_STORAGE_SANITIZERS)
  if(WIN32)
    target_compile_options(profile-storage PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
    target_link_options(profile-storage PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
  else()
    target_compile_options(profile-storage PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(profile-storage PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
  endif()
endif()
