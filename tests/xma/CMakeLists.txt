cmake_minimum_required(VERSION 3.25)
project(mw3_xma_lifecycle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rexglue 0.11 REQUIRED CONFIG)
add_executable(xma-lifecycle contracts.cpp)
target_link_libraries(xma-lifecycle PRIVATE rex::sdk)
rexglue_apply_target_settings(xma-lifecycle)
rexglue_configure_target(xma-lifecycle)
option(MW3_XMA_SANITIZERS "Instrument the original host harness" OFF)
if(MW3_XMA_SANITIZERS)
  if(WIN32)
    target_compile_options(xma-lifecycle PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
    target_link_options(xma-lifecycle PRIVATE -fsanitize=undefined -fsanitize-trap=all -fno-sanitize=vptr)
  else()
    target_compile_options(xma-lifecycle PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(xma-lifecycle PRIVATE -fsanitize=address,undefined -fno-sanitize=vptr)
  endif()
endif()
