cmake_minimum_required(VERSION 3.25)
project(mw3_image_probe LANGUAGES CXX)
option(MW3_SCAN_TESTS_ONLY "Build only game-free scan tests, without upstream dependencies" OFF)
option(MW3_SCAN_SANITIZERS "Instrument scan tests with address/undefined sanitizers" OFF)
add_executable(mw3-scan-tests scan_tests.cpp)
target_compile_features(mw3-scan-tests PRIVATE cxx_std_20)
if(MW3_SCAN_SANITIZERS AND NOT MSVC)
  target_compile_options(mw3-scan-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mw3-scan-tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME metadata-scans COMMAND mw3-scan-tests)
if(MW3_SCAN_TESTS_ONLY)
  return()
endif()
if(WIN32)
  message(FATAL_ERROR "This probe currently uses the canonical Ubuntu/WSL toolchain")
endif()
add_executable(mw3-image-probe image_probe.cpp)
target_compile_features(mw3-image-probe PRIVATE cxx_std_20)
target_include_directories(mw3-image-probe PRIVATE "${MW3_CPU_SOURCE}/XenonUtils"
  "${MW3_CPU_SOURCE}/thirdparty/simde" "${MW3_CPU_SOURCE}/thirdparty/disasm")
target_link_libraries(mw3-image-probe PRIVATE
  "${MW3_CPU_BUILD}/XenonUtils/libXenonUtils.a"
  "${MW3_CPU_BUILD}/thirdparty/disasm/libdisasm.a")
if(MW3_SCAN_SANITIZERS)
  target_compile_options(mw3-image-probe PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mw3-image-probe PRIVATE -fsanitize=address,undefined)
endif()
