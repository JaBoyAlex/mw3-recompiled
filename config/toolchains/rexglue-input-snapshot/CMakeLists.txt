cmake_minimum_required(VERSION 3.25)
project(mw3_input_snapshot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(rexglue 0.11 REQUIRED CONFIG)
find_package(SDL3 REQUIRED CONFIG)
add_library(rexhid-sdl3 SHARED src/hid/sdl3/sdl3_input_driver.cpp)
target_include_directories(rexhid-sdl3 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/src")
target_link_libraries(rexhid-sdl3 PRIVATE rex::base SDL3::SDL3)
set_target_properties(rexhid-sdl3 PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
rexglue_apply_target_settings(rexhid-sdl3)
rexglue_link_host_allocator(rexhid-sdl3)
if(UNIX AND NOT APPLE)
  rexglue_set_rpath(rexhid-sdl3 "$ORIGIN")
endif()
