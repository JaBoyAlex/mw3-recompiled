cmake_minimum_required(VERSION 3.25)
project(mw3_rexglue_experiment LANGUAGES CXX)
# This upstream SDK's public headers require C++23 (expected/byteswap).
# Isolated experiment only; the maintained game-free project remains C++20.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rexglue 0.11 REQUIRED CONFIG)
foreach(required MW3_GENERATED_DIR MW3_GENERATED_PREFIX MW3_PRIVATE_TRAPS)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing private experiment input: ${required}")
  endif()
endforeach()
include("${MW3_GENERATED_DIR}/sources.cmake")
add_library(guest OBJECT ${GENERATED_SOURCES})
target_link_libraries(guest PRIVATE rex::sdk)
target_include_directories(guest PRIVATE "${MW3_GENERATED_DIR}")
rexglue_apply_target_settings(guest)
rexglue_recomp_options(guest_options optimized)
target_compile_options(guest PRIVATE ${guest_options} -fwrapv)
target_precompile_headers(guest PRIVATE "${MW3_GENERATED_DIR}/${MW3_GENERATED_PREFIX}_pch.h")
add_executable(mw3-entry-probe main.cpp "${MW3_PRIVATE_TRAPS}")
target_compile_definitions(mw3-entry-probe PRIVATE MW3_GENERATED_INIT="${MW3_GENERATED_PREFIX}_init.h")
target_include_directories(mw3-entry-probe PRIVATE "${MW3_GENERATED_DIR}")
target_link_libraries(mw3-entry-probe PRIVATE guest rex::sdk)
rexglue_apply_target_settings(mw3-entry-probe)
rexglue_configure_target(mw3-entry-probe)
