cmake_minimum_required(VERSION 3.25)
project(mw3_native VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(mw3_xex STATIC src/xex.cpp)
target_include_directories(mw3_xex PUBLIC include)
target_compile_features(mw3_xex PUBLIC cxx_std_20)
set_target_properties(mw3_xex PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(mw3_xex PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(mw3_xex PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(MW3_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(MW3_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Use Clang or GCC for the sanitizer preset")
  endif()
  target_compile_options(mw3_xex PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mw3_xex PUBLIC -fsanitize=address,undefined)
endif()
add_executable(mw3-xex-inspect src/inspect.cpp)
target_link_libraries(mw3-xex-inspect PRIVATE mw3_xex)
if(BUILD_TESTING)
  add_executable(xex-tests tests/xex_tests.cpp)
  target_link_libraries(xex-tests PRIVATE mw3_xex)
  add_test(NAME xex-parser COMMAND xex-tests)
endif()
